# Sysadmin toolkit — the part that is always installed.
#
# Deliberately NOT everything a sysadmin might ever want. Kubernetes, Terraform
# and database clients are real tools for some people and dead weight for
# others, so they live in packages/optional-*.txt and are switched on in the
# settings or with --with <group>. A desktop that installs 40 things you never
# open is the thing we are trying not to build.

tmux
remmina
fuse-sshfs
rsync
rclone

# --- network and diagnostics ------------------------------------------------

sysstat
lsof
strace

# --- containers -------------------------------------------------------------
# podman only: it is what Fedora ships, runs rootless, and speaks the Docker
# API. Installing both podman and docker side by side causes more trouble than
# it solves.
podman
podman-compose
