# Desktop stack from the official Fedora 44 repositories.
# Everything here is native — see packages/copr-desktop.txt for the few
# pieces Fedora does not ship.

# --- login ------------------------------------------------------------------
sddm
sddm-wayland-generic

# --- bar, menus, notifications ---------------------------------------------
waybar
rofi
wlogout

# --- terminal and file manager ---------------------------------------------
kitty
nemo
nemo-fileroller

# --- screenshot / screen recording / clipboard ------------------------------
grim
slurp
wf-recorder
wl-clipboard
cliphist

# --- system controls --------------------------------------------------------
pavucontrol
blueman
network-manager-applet
nm-connection-editor
brightnessctl
playerctl
gammastep

# --- portals (file dialogs, screen sharing) ---------------------------------
xdg-desktop-portal
xdg-desktop-portal-gtk

# --- drives: cloud, network, phones -----------------------------------------
# gvfs mounts as your user, with no root and no fstab entry. That is why
# cifs-utils, nfs-utils and davfs2 are NOT here: they only add value for
# system-wide mounts, which this setup deliberately does not do.
gvfs
gvfs-smb
gvfs-nfs
gvfs-fuse
gvfs-mtp
gvfs-goa
fuse3

# --- accounts and secrets ---------------------------------------------------
# GOA supplies calendar, contacts and mail. It no longer supplies Google Drive
# FILES — GNOME 50 removed that; see docs/DRIVES.md. Files come from rclone.
gnome-online-accounts
gnome-online-accounts-gtk
gnome-keyring
libsecret
# seahorse (a GUI for the keyring) is deliberately left out: nothing here needs
# it, and an app you open once a year is exactly the kind of weight to avoid.

# --- calendar ---------------------------------------------------------------
# evolution-data-server is the sync engine behind all of it; gnome-calendar and
# the waybar module are just two views of the same data.
gnome-calendar

# --- theming ----------------------------------------------------------------
qt6ct
qt6-qtwayland
kvantum
papirus-icon-theme
# KDE's Breeze pointer. Ships two themes: breeze_cursors (dark) and
# Breeze_Light — there is no "Breeze_Dark", however much it sounds like it.
breeze-cursor-theme

# --- printing ---------------------------------------------------------------
# Fedora Server has no print stack at all, so a fresh install cannot print to
# anything. avahi (already running for mDNS) means a network printer is usually
# found without being told where it is.
cups
cups-filters
system-config-printer

# --- removable drives -------------------------------------------------------
# udisks is present but nothing acts on it, so plugging in a USB stick visibly
# does nothing. udiskie mounts it and says so.
udiskie

# --- fonts ------------------------------------------------------------------
# JetBrainsMono *Nerd Font* is not packaged; the plain family is, and
# lib/50-fonts-theme.sh installs the patched Nerd Font on top of it.
jetbrains-mono-fonts
rsms-inter-fonts
google-noto-color-emoji-fonts
# Without CJK coverage every Japanese, Chinese or Korean page in the browser is
# a wall of empty boxes. Costs disk once, fixes it permanently.
google-noto-sans-cjk-fonts

# --- runtime for the settings GUI and the bar popups ------------------------
python3-gobject
gtk4
libadwaita
# Anchors the bar popups to the top-right corner above tiled windows. Without
# it they would be ordinary windows that land wherever the layout decides.
gtk4-layer-shell

# --- fetch / monitor --------------------------------------------------------
btop
fastfetch

# WireGuard. NetworkManager speaks the protocol natively; these are the
# userspace tools for `wg show` and for generating keys.
wireguard-tools
